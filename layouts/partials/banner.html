{{/* banner.html – expects . to be a Page object */}} {{- with .Params.image -}} {{- $imgName := . -}} {{- $page := $ -}} {{- $img := $page.Resources.GetMatch $imgName -}} {{- if not $img -}} {{- $img = resources.Get (printf "banners/%s" $imgName) -}} {{- end -}} {{- if not $img -}} {{- $img = resources.Get "banners/default.png" -}} {{- end -}} {{- if not $img -}} {{- warnf "Banner %q not found as a page resource, in assets/banners/, or as assets/banners/default.png" $imgName -}} {{- else -}} {{- $banner := $img -}} {{- $srcset := slice -}} {{- if and (eq $img.ResourceType "image") (ne $img.MediaType.SubType "svg") -}} {{- /* Shown 16:9 (object-fit: cover) in a column at most 800px wide: crop to 16:9 and offer 1x / 1.5x / 2x widths, never upscaled. */ -}} {{- $base := math.Min 800 $img.Width | int -}} {{- $banner = $img.Fill (printf "%dx%d webp q82 Center" $base (div (mul $base 9) 16)) -}} {{- range slice 800 1200 1600 -}} {{- if le . $img.Width -}} {{- $v := $img.Fill (printf "%dx%d webp q82 Center" . (div (mul . 9) 16)) -}} {{- $srcset = $srcset | append (printf "%s %dw" $v.RelPermalink .) -}} {{- end -}} {{- end -}} {{- end -}}
